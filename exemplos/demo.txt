# Demonstração de alocação, diagnóstico e compactação
mkdir /sistema
mkdir /sistema/logs
create /sistema/logs/app.log --size 10KB
create /sistema/logs/encadeado.txt --size 4KB --alloc encadeada
fs_status --details
# A remoção de b deixa blocos livres separados e a alocação grande falha.
alloc contigua
create /a.txt --size 12KB
create /b.txt --size 12KB
create /c.txt --size 12KB
rm /b.txt
fs_status --details
create /grande.bin --size 16KB
compact
fs_status --details
create /grande.bin --size 16KB
fs_status --details
